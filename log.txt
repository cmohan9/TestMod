index=* (sourcetype="WinEventLog:Security" OR sourcetype="XmlWinEventLog:Security" OR source="WinEventLog:Security" OR source="XmlWinEventLog:Security") (EventCode IN (4728,4732,4756,4746,4751,4761,4729,4733,4757,4747,4752,4762) OR EventID IN (4728,4732,4756,4746,4751,4761,4729,4733,4757,4747,4752,4762)) "NOL-SEC-All_Member_Tatsuno"
| rex field=_raw "Subject:\s+Security ID:\s+[^\r\n]+\s+Account Name:\s+(?<raw_actor>\S+)\s+Account Domain:\s+(?<raw_actor_dom>\S+)"
| rex field=_raw "Member:\s+Security ID:\s+(?<raw_sid>\S+)\s+Account Name:\s+(?<raw_member>[^\r\n]+)"
| rex field=_raw "Group:\s+Security ID:\s+[^\r\n]+\s+Group Name:\s+(?<raw_group>[^\r\n]+)"
| eval Event=coalesce(EventCode, EventID)
| eval Action=if(Event IN ("4728","4732","4756","4746","4751","4761"),"ADDED","REMOVED")
| eval Changed_By=coalesce(SubjectUserName, raw_actor)
| eval Actor_Domain=coalesce(SubjectDomainName, raw_actor_dom)
| eval Member_DN=coalesce(MemberName, raw_member)
| eval Member_SID=coalesce(MemberSid, raw_sid)
| eval Group_Name=coalesce(TargetUserName, raw_group)
| search Group_Name="NOL-SEC-All_Member_Tatsuno"
| eval Time=strftime(_time,"%Y-%m-%d %H:%M:%S")
| table Time Action Group_Name Member_DN Member_SID Changed_By Actor_Domain Event host
| sort 0 Time
